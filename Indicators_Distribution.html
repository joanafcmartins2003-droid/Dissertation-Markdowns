    cols = -Age_Group,

  mutate(

    CountryBirth = as.character(CountryBirth),

total_pop_official <- sum(official_long$Pop, na.rm = TRUE)

pop_country_ref <- official_long %>%
  group_by(CountryBirth) %>%
  summarise(Pop = sum(Pop, na.rm = TRUE), .groups = "drop") %>%
  mutate(
    Oficial_Pop_pp = (Pop / total_pop_official) * 100, 
    Dimension = "CountryBirth", 
    Category = CountryBirth
  )

pop_crossed_ref <- official_long %>%
  mutate(
    Oficial_Pop_pp = (Pop / total_pop_official) * 100, 
    Dimension = "CountryBirth x Age", 
    Category = paste(CountryBirth, Age_Group, sep = "_")
  )

official_ref <- bind_rows(
  pop_country_ref %>% select(Dimension, Category, Oficial_Pop_pp),
  pop_crossed_ref %>% select(Dimension, Category, Oficial_Pop_pp)
)

GEM_weights <- GEM %>%
  filter(!is.na(Age), !is.na(CountryBirth)) %>%
  mutate(
    Country_clean = as.character(CountryBirth),
    Age_clean = as.character(Age),
    `1. GEM Unweighted`      = 1,
    `2. Current Weight`     = Weight18_64,
    `3. Shared Weight`       = shared_weight
  ) %>%
  select(Country_clean, Age_clean, starts_with("1."), starts_with("2."), starts_with("3.")) %>%
  pivot_longer(cols = c(`1. GEM Unweighted`, `2. Current Weight`, `3. Shared Weight`), 
               names_to = "Model", values_to = "W")

gem_dist_country <- GEM_weights %>%
  group_by(Model, Category = Country_clean) %>%
  summarise(sum_w = sum(W, na.rm = TRUE), .groups = "drop_last") %>%
  mutate(Model_Percentage = (sum_w / sum(sum_w)) * 100, Dimension = "CountryBirth")

gem_dist_crossed <- GEM_weights %>%
  mutate(Category = paste(Country_clean, Age_clean, sep = "_")) %>%
  group_by(Model, Category) %>%
  summarise(sum_w = sum(W, na.rm = TRUE), .groups = "drop_last") %>%
  mutate(Model_Percentage = (sum_w / sum(sum_w)) * 100, Dimension = "CountryBirth x Age")


comparison_df <- bind_rows(gem_dist_country, gem_dist_crossed) %>%
  left_join(official_ref, by = c("Dimension", "Category")) %>%
  filter(!is.na(Oficial_Pop_pp)) %>% 
  mutate(Abs_Error = abs(Model_Percentage - Oficial_Pop_pp))

final_tab <- comparison_df %>%

  left_join(official_ref, by = c("Dimension", "Category")) %>%
  rename(`Official (%)` = Oficial_Pop_pp) %>%
  mutate(across(where(is.numeric), ~ round(., 2)))

winners_by_group <- comparison_df %>%

  select(Dimension, Category, `Official (%)` = Oficial_Pop_pp,

ranking_by_dimension <- comparison_df %>%
